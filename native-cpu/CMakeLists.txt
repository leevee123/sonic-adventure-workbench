cmake_minimum_required(VERSION 3.16)
project(SonicAdventureCPU C)
set(CMAKE_C_STANDARD 11)
file(GLOB DOL_CHUNKS "${CMAKE_CURRENT_SOURCE_DIR}/../recompiled/dol/generated/chunks/*.c")
add_library(sadx_cpu STATIC ${DOL_CHUNKS})
target_include_directories(sadx_cpu PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../toolchain/DolRecomp/src")
target_compile_definitions(sadx_cpu PRIVATE DOLRECOMP_CPU_HEADER="host_cpu.h")
add_executable(retail_cpu_smoke retail_cpu_smoke.c "${CMAKE_CURRENT_SOURCE_DIR}/../../toolchain/DolRecomp/src/cpu/cpu.c")
target_include_directories(retail_cpu_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../toolchain/DolRecomp/src")
target_compile_definitions(retail_cpu_smoke PRIVATE DOLRECOMP_CPU_HEADER="host_cpu.h")
target_link_libraries(retail_cpu_smoke PRIVATE sadx_cpu)
